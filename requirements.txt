fastapi

uvicorn[standard]


#########################################
# LangChain – Núcleo y módulos esenciales
#########################################

# Base interna de LangChain (PromptTemplate, OutputParser, Runnables)
langchain-core

# Framework principal de LangChain (chains, tools, agentes, etc.)
langchain

# Herramientas de división de texto (RecursiveTextSplitter, etc.)
langchain-text-splitters

# Conectores para modelos OpenAI (también funciona con OpenRouter)
langchain-openai

# Conectores para modelos Gemini (Google AI)
langchain-google-genai

# Integraciones comunitarias (loaders de documentos, utils, vectorstores, etc.)
langchain-community


#########################################
# Modelos LLM y Embeddings
#########################################

# Cliente OpenAI moderno (requerido por OpenRouter y OpenAI)
openai

# Embeddings y modelos locales (MiniLM, MPNet, etc.)
sentence-transformers


#########################################
# Vector Database para RAG
#########################################

# Base de datos de vectores local y ligera (ideal para prototipos y entornos locales)
chromadb


#########################################
# Scraping Web
#########################################

# Cliente HTTP básico para hacer solicitudes web
requests

# Librería para parseo de HTML (web scraping)
beautifulsoup4


#########################################
# Utilidades generales y helpers
#########################################

# Cargar variables desde .env (API keys, configuración)
python-dotenv

# Validación de datos (Pydantic v2 — requerido por FastAPI y LangChain)
pydantic

# Cliente HTTP asincrónico usado por muchos SDKs modernos
httpx

# Validación de emails (evita warnings en FastAPI/Pydantic)
email-validator

langgraph

langsmith

langgraph-checkpoint
